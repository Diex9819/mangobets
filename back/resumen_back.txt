import 'dotenv/config';
import mongoose from 'mongoose';
import { cargarUriLocal } from '../src/utils/dbGuard.js';

const NOMBRE_INDICE_OBSOLETO = 'uniq_pago_finanza';

function log(...args) {
  console.log('[fix-indice-pagonomina]', ...args);
}

function error(...args) {
  console.error('[fix-indice-pagonomina][ERROR]', ...args);
}

function buscarIndicesFinanzaId(indices) {
  return indices.filter((idx) => idx.name === NOMBRE_INDICE_OBSOLETO || Object.keys(idx.key).includes('finanzaId'));
}

async function main() {
  const { uri, db } = cargarUriLocal();
  log(`Conectando a MongoDB LOCAL (host localhost, db=${db})`);
  await mongoose.connect(uri);
  log('Conexión establecida.');

  const col = mongoose.connection.collection('pagonominas');
  const indicesActuales = await col.indexes();

  const obsoletos = buscarIndicesFinanzaId(indicesActuales);
  if (obsoletos.length === 0) {
    log('No se encontró ningún índice obsoleto de finanzaId. Nada que hacer.');
    return;
  }

  for (const idx of obsoletos) {
    log(`Índice obsoleto encontrado: "${idx.name}" (key=${JSON.stringify(idx.key)}, unique=${!!idx.unique}). Eliminándolo...`);
    await col.dropIndex(idx.name);
    log(`Índice "${idx.name}" eliminado correctamente.`);
  }

  const indicesRestantes = await col.indexes();
  log('Índices vigentes de la colección pagonominas:');
  for (const idx of indicesRestantes) {
    log(`  ${idx.name}: key=${JSON.stringify(idx.key)}${idx.unique ? ' unique' : ''}${idx.partialFilterExpression ? ` partial=${JSON.stringify(idx.partialFilterExpression)}` : ''}`);
  }

  if (buscarIndicesFinanzaId(indicesRestantes).length > 0) {
    throw new Error(`Verificación fallida: aún existe un índice con clave finanzaId en ${indicesRestantes.map((i) => i.name).join(', ')}`);
  }
  log('Verificación OK: no quedan índices de finanzaId en pagonominas.');
}

main()
  .catch((err) => {
    error(err.message);
    process.exitCode = 1;
  })
  .finally(async () => {
    await mongoose.connection.close();
  });

import 'dotenv/config';
import mongoose from 'mongoose';
import { cargarUriLocal } from '../src/utils/dbGuard.js';

// 📅 Migración de índices para PAGOS PERIÓDICOS PARCIALES.
//
// 1) Elimina el índice único (empleado + periodo) que impedía registrar más de
//    un abono PERIODICO vigente en el mismo periodo.
// 2) Verifica/crea el índice de protección de duplicados:
//    (empleado, periodo, fechaPago, monto) único y parcial, que permite varios
//    abonos parciales pero bloquea un abono IDÉNTICO (mismo día y monto).

const INDICE_OBSOLETO = 'uniq_empleado_periodo_periodico';
const INDICE_DEDUP = 'uniq_periodo_pago';

function log(...args) {
  console.log('[fix-indice-pagonomina-parcial]', ...args);
}

function error(...args) {
  console.error('[fix-indice-pagonomina-parcial][ERROR]', ...args);
}

async function main() {
  const { uri, db } = cargarUriLocal();
  log(`Conectando a MongoDB LOCAL (host localhost, db=${db})`);
  await mongoose.connect(uri);
  log('Conexión establecida.');

  const col = mongoose.connection.collection('pagonominas');
  const indicesActuales = await col.indexes();

  // 1) Eliminar el índice único que bloqueaba los pagos parciales
  const obsoleto = indicesActuales.find((idx) => idx.name === INDICE_OBSOLETO);
  if (obsoleto) {
    log(`Eliminando índice obsoleto "${INDICE_OBSOLETO}" (key=${JSON.stringify(obsoleto.key)}, unique=${!!obsoleto.unique})...`);
    await col.dropIndex(INDICE_OBSOLETO);
    log(`Índice "${INDICE_OBSOLETO}" eliminado correctamente.`);
  } else {
    log(`El índice "${INDICE_OBSOLETO}" no existe. Nada que eliminar.`);
  }

  // 2) Índice de protección de duplicados (permite parciales, bloquea idénticos)
  const yaExisteDedup = indicesActuales.some((idx) => idx.name === INDICE_DEDUP);
  if (yaExisteDedup) {
    log(`El índice "${INDICE_DEDUP}" ya existe. No se crea de nuevo.`);
  } else {
    await col.createIndex(
      { empleado: 1, periodo: 1, fechaPago: 1, monto: 1 },
      { unique: true, partialFilterExpression: { tipoPago: 'PERIODICO', anulado: false }, name: INDICE_DEDUP }
    );
    log(`Índice "${INDICE_DEDUP}" creado correctamente.`);
  }

  const indicesRestantes = await col.indexes();
  log('Índices vigentes de la colección pagonominas:');
  for (const idx of indicesRestantes) {
    log(`  ${idx.name}: key=${JSON.stringify(idx.key)}${idx.unique ? ' unique' : ''}${idx.partialFilterExpression ? ` partial=${JSON.stringify(idx.partialFilterExpression)}` : ''}`);
  }

  if (indicesRestantes.some((idx) => idx.name === INDICE_OBSOLETO)) {
    throw new Error(`Verificación fallida: aún existe el índice "${INDICE_OBSOLETO}" en pagonominas.`);
  }
  log('Verificación OK: sin índice (empleado+periodo); con índice de duplicados vigente.');
}

main()
  .catch((err) => {
    error(err.message);
    process.exitCode = 1;
  })
  .finally(async () => {
    await mongoose.connection.close();
  });

import 'dotenv/config';

const {
  MONGODB_URI,
  ADMIN_NAME = 'Administrador',
  ADMIN_USERNAME,
  ADMIN_PASSWORD,
} = process.env;

async function seedAdmin() {
  if (!MONGODB_URI) {
    throw new Error('MONGODB_URI is required');
  }

  if (!ADMIN_USERNAME || !ADMIN_PASSWORD) {
    throw new Error('ADMIN_USERNAME and ADMIN_PASSWORD are required');
  }

  if (ADMIN_PASSWORD.length < 8) {
    throw new Error('ADMIN_PASSWORD must be at least 8 characters');
  }

  await mongoose.connect(MONGODB_URI);

  const existingAdmin = await User.findOne({ role: 'ADMIN' });
  if (existingAdmin) {
    console.log(`Admin already exists: ${existingAdmin.username}`);
    return;
  }

  const existingUsername = await User.findOne({ username: ADMIN_USERNAME });
  if (existingUsername) {
    throw new Error(`Username already exists: ${ADMIN_USERNAME}`);
  }

  const admin = await User.create({
    name: ADMIN_NAME,
    username: ADMIN_USERNAME,
    password: ADMIN_PASSWORD,
    role: 'ADMIN',
  });

  console.log(`Admin created: ${admin.username}`);
}

seedAdmin()
  .catch((error) => {
    console.error(error.message);
    process.exitCode = 1;
  })
  .finally(async () => {
    await mongoose.connection.close();

const MS_DIA = 86400000;

export function mulberry32(seed) {
  let a = seed >>> 0;
  return function () {
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

export function randInt(rng, min, max) {
  return min + Math.floor(rng() * (max - min + 1));
}

export function chance(rng, p) {
  return rng() < p;
}

export function pick(rng, arr) {
  return arr[Math.floor(rng() * arr.length)];
}

export function weightedPick(rng, items, weights) {
  if (items.length !== weights.length) throw new Error('longitudes distintas');
  let suma = 0;
  for (const w of weights) suma += w;
  let objetivo = rng() * suma;
  for (let i = 0; i < items.length; i++) {
    objetivo -= weights[i];
    if (objetivo <= 0) return items[i];
  }
  return items[items.length - 1];
}

export function round2(n) {
  return Math.round(n * 100) / 100;
}

export function startOfDay(date) {
  return new Date(Date.UTC(date.getUTCFullYear(), date.getUTCMonth(), date.getUTCDate()));
}

export function addDays(date, days) {
  return new Date(date.getTime() + days * MS_DIA);
}

export function fmtFecha(date) {
  return date.toISOString().slice(0, 10);
}

export function hoyUtc() {

function armarLinea({ rng, productos, perfil, nProd, dia, conBotella = false }) {
  const productosLinea = [];
  let incluirBotella = false;
  if (conBotella && chance(rng, 0.25)) incluirBotella = true;

  for (let j = 0; j < nProd; j++) {
    let prod;
    if (incluirBotella && j === 0) {
      const botellas = productos.filter((p) => p.categoria === CATEGORIA_BOTELLA);
      prod = botellas.length ? botellas[randInt(rng, 0, botellas.length - 1)] : productos[0];
    } else {
      prod = weightedPick(rng, productos, productos.map((p) => p.peso));
    }
    if (prod.stock === 0 && dia >= 25) {
      j -= 1;
      continue;
    }
    const yaExiste = productosLinea.find((linea) => linea.productId === prod._id);
    if (yaExiste) {
      yaExiste.cantidad += 1;
    } else {
      productosLinea.push({
        productId: prod._id,
        nombre: prod.name,
        precio: prod.price,
        cantidad: 1,
      });
    }
  }

  return productosLinea;
}

  for (let dia = 0; dia < totalDias; dia++) {
    const fecha = addDays(inicio, dia);
    const diaSemana = fecha.getUTCDay();
    const factor = FACTOR_DIA[diaSemana];

    if (chance(rng, 0.04) && diaSemana >= 1 && diaSemana <= 3) {
      continue; // día sin ventas (lunes-miércoles principalmente)
    }

    const n = Math.round(9 * factor * (0.8 + rng() * 0.4));
    if (n <= 0) continue;
    diasAbiertosConVentas.push(dia);

    for (let i = 0; i < n; i++) {
      const perfil = weightedPick(rng, PERFIL_TICKET, PERFIL_TICKET.map((p) => p.prob));
      const nProd = randInt(rng, perfil.minP, perfil.maxP);
      const esGrupo = perfil.tipo === 'grupo';

      const productosLinea = armarLinea({
        rng, productos, perfil, nProd, dia,
        conBotella: perfil.conBotella,
      });

      const [hora, minuto] = escogerHorario(rng);
      const createdAt = new Date(Date.UTC(fecha.getUTCFullYear(), fecha.getUTCMonth(), fecha.getUTCDate(), hora, minuto));

      const total = round2(productosLinea.reduce((s, p) => s + p.precio * p.cantidad, 0));

      const metodoPago = escogerMetodoPago(rng);
      const mesa = elegirMesa(rng, mesas, esGrupo);

      planeadas.push({
        doc: {
          mesaId: mesa._id,
          productos: productosLinea,
          total,
          estado: 'CERRADA',
          corteId: null,
          metodoPago,
          esDemo: true,
          createdAt,
          updatedAt: createdAt,
        },
        dia,
      });
    }

    // Operaciones de MADRUGADA (00:00-00:59): pertenecen a la MISMA caja/sesión
    // que la noche anterior (el cierre ocurre entre la 1:00 y las 4:00).
    const nMadrugada = randInt(rng, 0, 3);
    for (let m = 0; m < nMadrugada; m++) {
      const perfil = pick(rng, PERFIL_TICKET);
      const nProd = randInt(rng, perfil.minP, perfil.maxP);
      const productosLinea = armarLinea({
        rng, productos, perfil, nProd, dia,
        conBotella: perfil.tipo === 'grupo',
      });
      if (productosLinea.length === 0) continue;

      const siguienteDia = addDays(fecha, 1);
      const createdAt = new Date(Date.UTC(
        siguienteDia.getUTCFullYear(),
        siguienteDia.getUTCMonth(),
        siguienteDia.getUTCDate(),
        0,
        randInt(rng, 5, 55)
      ));

      const total = round2(productosLinea.reduce((s, p) => s + p.precio * p.cantidad, 0));
      const mesa = elegirMesa(rng, mesas, perfil.tipo === 'grupo');

      planeadas.push({
        doc: {
          mesaId: mesa._id,
          productos: productosLinea,
          total,
          estado: 'CERRADA',
          corteId: null,
          metodoPago: escogerMetodoPago(rng),
          esDemo: true,
          createdAt,
          updatedAt: createdAt,
        },
        dia,
      });
    }
  }

  return { planeadas, diasAbiertosConVentas };
}

export function escogerMetodoPago(rng) {
  const r = rng();
  if (r < 0.58) return 'EFECTIVO';
  if (r < 0.84) return 'TARJETA';
  if (r < 0.92) return 'TRANSFERENCIA';
  return 'DEUDA';
}

export function construirCorte({ diaDocs, usuarioId, fechaInicio = null, fechaFin = null }) {
  let totalVentas = 0;
  let totalEfectivo = 0;
  let totalTarjeta = 0;
  let totalTransferencia = 0;
  let totalDeuda = 0;

  for (const orden of diaDocs) {
    totalVentas += orden.total;
    if (orden.metodoPago === 'EFECTIVO') totalEfectivo += orden.total;
    else if (orden.metodoPago === 'TARJETA') totalTarjeta += orden.total;
    else if (orden.metodoPago === 'TRANSFERENCIA') totalTransferencia += orden.total;
    else if (orden.metodoPago === 'DEUDA') totalDeuda += orden.total;
  }

  const times = diaDocs.map((o) => o.createdAt.getTime()).sort((a, b) => a - b);
  const doc = {
    fechaInicio: fechaInicio || (times.length ? new Date(times[0]) : new Date()),
    fechaFin: fechaFin || (times.length ? new Date(times[times.length - 1]) : new Date()),
    totalVentas: round2(totalVentas),
    totalEfectivo: round2(totalEfectivo),
    totalTarjeta: round2(totalTarjeta),
    totalTransferencia: round2(totalTransferencia),
    totalDeuda: round2(totalDeuda),
    cantidadTickets: diaDocs.length,
    tickets: diaDocs.map((o) => o._id),
    usuario: usuarioId,
    esDemo: true,
  };

  return { doc, orderIds: diaDocs.map((o) => o._id) };
}

export function planificarCierres({ ordenesPorDia, totalDias, diasSinCorte, rng }) {
  const sesiones = [];
  let ultimoFin = null;

  for (let dia = 0; dia < totalDias; dia++) {
    const docs = ordenesPorDia.get(dia) || [];
    if (docs.length === 0) continue; // día sin ventas: no hay sesión ni cierre

    if (dia >= totalDias - diasSinCorte) break; // últimos días: caja sigue abierta

    const cierreDia = addDays(startOfDay(docs[0].createdAt), 1);
    const fechaFin = new Date(Date.UTC(
      cierreDia.getUTCFullYear(),
      cierreDia.getUTCMonth(),
      cierreDia.getUTCDate(),
      randInt(rng, 1, 4),
      randInt(rng, 0, 59)
    ));

    sesiones.push({
      dia,
      diaDocs: docs,
      fechaInicio: ultimoFin || docs[0].createdAt,
      fechaFin,
    });
    ultimoFin = fechaFin;
  }

  return sesiones;
}

export function siguienteVencimiento({ periodicidad, base, diaPago = null, diaSemana = null }) {
  if (periodicidad === 'MENSUAL') {
    const siguiente = new Date(Date.UTC(base.getUTCFullYear(), base.getUTCMonth() + 1, 1));
    const ultimo = new Date(Date.UTC(siguiente.getUTCFullYear(), siguiente.getUTCMonth() + 1, 0)).getUTCDate();
    return new Date(Date.UTC(siguiente.getUTCFullYear(), siguiente.getUTCMonth(), Math.min(diaPago || 1, ultimo)));
  }
  if (periodicidad === 'SEMANAL') {
    return addDays(base, 7);
  }
  throw new Error(`periodicidad no soportada en demo: ${periodicidad}`);
}

function ultimoDiaMes(date) {
  return new Date(Date.UTC(date.getUTCFullYear(), date.getUTCMonth() + 1, 0)).getUTCDate();
}

function primerVencimientoSemanal(inicio, diaSemana) {
  const delta = (diaSemana - inicio.getUTCDay() + 7) % 7;
  return addDays(startOfDay(inicio), delta);
}

function primerVencimientoMensual(inicio, diaPago) {
  const cand = new Date(Date.UTC(inicio.getUTCFullYear(), inicio.getUTCMonth(), Math.min(diaPago, ultimoDiaMes(inicio))));
  if (cand.getTime() >= inicio.getTime()) return cand;
  const sigMes = new Date(Date.UTC(inicio.getUTCFullYear(), inicio.getUTCMonth() + 1, 1));
  return new Date(Date.UTC(sigMes.getUTCFullYear(), sigMes.getUTCMonth(), Math.min(diaPago, ultimoDiaMes(sigMes))));
}

        name: user.name,

      success: false,
      error: 'No se pudo crear el usuario'
    });
  }

// 🔐 LOGIN
export const login = async (req, res) => {
  try {
    const { username, password } = req.body;

    if (!username || !password) {
      return res.status(400).json({
        success: false,
        message: 'Credenciales inválidas'
      });
    }

    // 🔥 importante traer password
    const user = await User.findOne({ username }).select('+password');

    if (!user) {
      return res.status(400).json({
        success: false,
        message: 'Credenciales inválidas'
      });
    }

    const isMatch = await user.comparePassword(password);

    if (!isMatch) {
      return res.status(400).json({
        success: false,
        message: 'Credenciales inválidas'
      });
    }

    res.json({
      success: true,
      token: generateToken(user._id),
      user: {
        _id: user._id,
        name: user.name,
        username: user.username,
        role: user.role
      }
    });

  } catch (error) {
    console.error('Login error:', error);
    res.status(500).json({
      success: false,
      error: 'Error al iniciar sesión'
    });
  }
};

// Cierra el periodo ABIERTO, guarda el snapshot con totales del backend y deja
// inmediatamente un nuevo periodo abierto que arranca EXACTAMENTE en la fechaFin.
export const cerrarPeriodo = async (req, res) => {
  try {
    // 1. Período abierto actual (con bootstrap si nunca se ha inicializado)
    const abierto = await getPeriodoAbierto();

    // 2. 🔒 Cierre atómico anti doble cierre: el filtro exige estado ABIERTO,
    //    así que SOLO una solicitud logra el salto ABIERTO → CERRADO.
    const cerrado = await CierreFinanciero.findOneAndUpdate(
      { _id: abierto._id, estado: 'ABIERTO', esAbierto: true },
      {
        $set: {
          estado: 'CERRADO',
          esAbierto: false,
          fechaFin: new Date(),
          usuario: req.user?.name || null,
          usuarioId: req.user?._id || null,
        },
      },
      { new: true }
    );

    if (!cerrado) {
      return res.status(409).json({
        error: 'Este periodo ya fue cerrado por otra solicitud (no se permite un cierre simultáneo).',
      });
    }

    // 3. 🧮 Totales calculados por BACKEND (nunca confiar en lo que envía Angular)
    const totales = await calcularTotalesPeriodo(cerrado._id);
    cerrado.ingresos = totales.ingresos;

    cerrado.inversiones = totales.inversiones;
    cerrado.resultado = totales.resultado;
    cerrado.cantidadMovimientos = totales.cantidadMovimientos;

    // 4. 🔢 Número secuencial del historial (CIERRE #1, #2, ...)
    const ultimo = await CierreFinanciero.findOne({ estado: 'CERRADO' })
      .sort({ numero: -1 })
      .select('numero')
      .lean();
    cerrado.numero = (ultimo?.numero || 0) + 1;

    await cerrado.save();

    // 5. Nuevo periodo abierto: fechaInicio === fechaFin del cierre (continuidad
    //    exacta, sin saltos ni solapamientos). getPeriodoAbierto es idempotente,
    //    así que también absorbe cualquier creación concurrente de periodo.
    const nuevo = await getPeriodoAbierto();
    const totalesNuevo = await calcularTotalesPeriodo(nuevo._id);

// Normaliza 'YYYY-MM-DD' a medianoche (UTC) para evitar bugs de zona horaria
function fechaInicioDia(fechaStr) {
  return new Date(`${fechaStr}T00:00:00.000Z`);

function esFechaValida(fechaStr) {
  return /^\d{4}-\d{2}-\d{2}$/.test(fechaStr);
}

// GET /api/finanzas?fecha=YYYY-MM-DD
export const getDia = async (req, res) => {
  try {
    const { fecha } = req.query;

    if (!fecha || !esFechaValida(fecha)) {
      return res.status(400).json({ error: 'Fecha inválida, usa formato YYYY-MM-DD' });
    }

    // 💼 El periodo ABIERTO actual acota la vista al periodo vigente, así los
    // movimientos del periodo recién cerrado (aunque compartan fecha) ya no aparecen.
    const periodoActual = await getPeriodoAbierto();

    const inicio = fechaInicioDia(fecha);
    const fin = new Date(inicio.getTime() + 86400000);

    const registros = await Finanza.find({
      fecha: { $gte: inicio, $lt: fin },
      cierreFinancieroId: periodoActual._id,
    })
      .populate('usuario', 'name username')
      .sort({ createdAt: -1 });

    const ingresos = registros.filter((r) => r.tipo === 'INGRESO');
    const gastos = registros.filter((r) => r.tipo === 'GASTO');
    const inversiones = registros.filter((r) => r.tipo === 'INVERSION');

    const sumar = (arr) => arr.reduce((acc, r) => acc + r.monto, 0);

    const totalIngresos = sumar(ingresos);
    const totalGastos = sumar(gastos);
    const totalInversiones = sumar(inversiones);

    res.json({
      fecha,
      ingresos,
      gastos,
      inversiones,
      resumen: {
        totalIngresos,
        totalGastos,
        totalInversiones,
        resultado: totalIngresos - totalGastos - totalInversiones,
      },
    });
  } catch (err) {
    res.status(500).json({ error: 'Error al obtener finanzas del día' });
  }

      );
      if (index < 0) {

      }

      // de periodos anteriores que permanecen archivados en el historial.
      const montoPagado = round2(
        Math.max(0, (Number(rec.montoPagado) || 0) + Number(finanza.monto) - montoAbonoAnterior)
      );

      rec.montoPagado = montoPagado;

      // 🏷️ Reasigna la referencia superior al último abono vigente.

      if (ultimo) {

            : rec.monto != null
            ? Number(rec.monto)
            : null;

        }
      }

        monto: h.monto,
        finanzaId: h.finanzaId,
      }));

    }

function camposValidos(body) {
  return {
    tipo: body.tipo,
    fecha: fechaInicioDia(body.fecha),
    concepto: body.concepto?.trim(),
    monto: Number(body.monto),
    metodoPago: body.tipo !== 'INVERSION' ? body.metodoPago : undefined,
    categoria: body.tipo === 'GASTO' ? body.categoria?.trim() : undefined,
    nota: body.nota?.trim() || '',
  };
}

function validar(body) {
  const errores = [];

  if (!TIPOS.includes(body.tipo)) {
    errores.push({ field: 'tipo', message: 'tipo debe ser INGRESO, GASTO o INVERSION' });
  }

  if (!body.fecha || !esFechaValida(body.fecha)) {
    errores.push({ field: 'fecha', message: 'fecha es requerida en formato YYYY-MM-DD' });
  }

  if (!body.concepto || !String(body.concepto).trim()) {
    errores.push({ field: 'concepto', message: 'concepto es requerido' });
  }

  const monto = Number(body.monto);
  if (!monto || monto <= 0) {
    errores.push({ field: 'monto', message: 'monto debe ser un número mayor a 0' });
  }

  if (body.tipo === 'INGRESO' || body.tipo === 'GASTO') {
    if (!METODOS_PAGO.includes(body.metodoPago)) {
      errores.push({ field: 'metodoPago', message: 'metodoPago es requerido' });
    }
  }

  if (body.tipo === 'GASTO' && !String(body.categoria || '').trim()) {
    errores.push({ field: 'categoria', message: 'categoria es requerida' });
  }

  return errores;
}

const ESTADOS = ['PENDIENTE', 'PAGADO', 'INACTIVO'];
const PERIODICIDADES = ['SEMANAL', 'QUINCENAL', 'MENSUAL', 'BIMESTRAL', 'TRIMESTRAL', 'SEMESTRAL', 'ANUAL', 'OTRA'];
const METODOS_PAGO = ['EFECTIVO', 'TARJETA', 'TRANSFERENCIA', 'OTRO'];
const CATEGORIA_GASTO_FIJO = 'Gastos fijos';

// Normaliza 'YYYY-MM-DD' a medianoche (UTC) igual que Finanzas
function fechaInicioDia(fechaStr) {
  return new Date(`${fechaStr}T00:00:00.000Z`);
}

function esFechaValida(fechaStr) {
  return /^\d{4}-\d{2}-\d{2}$/.test(fechaStr);
}

function hoyUTC() {
  const now = new Date();
  return new Date(Date.UTC(now.getUTCFullYear(), now.getUTCMonth(), now.getUTCDate()));
}

function round2(n) {
  return Math.round((Number(n) + Number.EPSILON) * 100) / 100;
}

// 🔢 Suma meses en UTC evitando problèmes con días 29/30/31
function addMonths(date, months) {
  const year = date.getUTCFullYear();
  const month = date.getUTCMonth() + months;
  const day = date.getUTCDate();

  const targetYear = year + Math.floor(month / 12);
  const targetMonth = ((month % 12) + 12) % 12;
  const lastDay = new Date(Date.UTC(targetYear, targetMonth + 1, 0)).getUTCDate();

  return new Date(Date.UTC(targetYear, targetMonth, Math.min(day, lastDay)));
}

function addDays(date, days) {
  return new Date(date.getTime() + days * 86400000);
}

// 🔢 Calcula el siguiente vencimiento según la periodicidad
function siguienteVencimiento(periodicidad, base, nuevaFecha) {
  switch (periodicidad) {
    case 'SEMANAL':
      return addDays(base, 7);
    case 'QUINCENAL':
      return addDays(base, 15);
    case 'MENSUAL':
      return addMonths(base, 1);
    case 'BIMESTRAL':
      return addMonths(base, 2);
    case 'TRIMESTRAL':
      return addMonths(base, 3);
    case 'SEMESTRAL':
      return addMonths(base, 6);
    case 'ANUAL':
      return addMonths(base, 12);
    case 'OTRA':
      return nuevaFecha ? fechaInicioDia(nuevaFecha) : base;
    default:
      return base;
  }
}

// POST /api/gastos-recurrentes
export const crear = async (req, res) => {
  try {
    const errores = validar(req.body);
    if (errores.length > 0) {
      return res.status(422).json({ error: 'Validation failed', details: errores });
    }

    const rec = await GastoRecurrente.create({
      nombre: req.body.nombre.trim(),
      monto: req.body.montoVariable ? null : Number(req.body.monto),
      montoVariable: !!req.body.montoVariable,
      periodicidad: req.body.periodicidad,
      proximoPago: fechaInicioDia(req.body.proximoPago),
      nota: req.body.nota?.trim() || '',
      estado: 'PENDIENTE',
      usuario: req.user._id,
    });

    res.status(201).json(rec);
  } catch (err) {
    res.status(500).json({ error: 'Error al crear gasto recurrente' });
  }
};

// PUT /api/gastos-recurrentes/:id
export const actualizar = async (req, res) => {
  try {
    const rec = await GastoRecurrente.findById(req.params.id);
    if (!rec) {
      return res.status(404).json({ error: 'Gasto recurrente no encontrado', id: req.params.id });
    }

    const errores = validar(req.body);
    if (errores.length > 0) {
      return res.status(422).json({ error: 'Validation failed', details: errores });
    }

    if (req.body.estado && req.body.estado === 'PAGADO') {
      return res.status(400).json({ error: 'No puedes marcar PAGADO manualmente, usa el endpoint de pago' });
    }
    if (req.body.estado && !ESTADOS.includes(req.body.estado)) {
      return res.status(422).json({ error: 'estado inválido' });
    }

    if (req.body.nombre) rec.nombre = req.body.nombre.trim();
    if (req.body.montoVariable !== undefined) rec.montoVariable = !!req.body.montoVariable;
    if (req.body.monto !== undefined) {
      // 💰 La variable admite referencia opcional; null enmarca "se captura al registrar el recibo"
      rec.monto = req.body.monto === null || req.body.monto === '' ? null : Number(req.body.monto);
    }
    if (req.body.periodicidad) rec.periodicidad = req.body.periodicidad;
    if (req.body.proximoPago) rec.proximoPago = fechaInicioDia(req.body.proximoPago);
    if (req.body.nota !== undefined) rec.nota = req.body.nota?.trim() || '';
    if (req.body.estado) rec.estado = req.body.estado; // solo PENDIENTE o INACTIVO

    await rec.save();
    res.json(rec);
  } catch (err) {
    res.status(500).json({ error: 'Error al actualizar gasto recurrente' });
  }
};

  }
};

// POST /api/gastos-recurrentes/:id/pagar
export const pagar = async (req, res) => {
  try {
    const { montoPagado, metodoPago, fechaPago, montoRecibo } = req.body;

    if (!METODOS_PAGO.includes(metodoPago)) {
      return res.status(422).json({ error: 'metodoPago es requerido' });
    }

    const pago = round2(Number(montoPagado));
    if (!pago || pago <= 0) {
      return res.status(422).json({ error: 'montoPagado es requerido y debe ser mayor a 0' });
    }

    const fechaPagoISO = fechaPago && esFechaValida(fechaPago) ? fechaInicioDia(fechaPago) : hoyUTC();

    const rec = await GastoRecurrente.findById(req.params.id);
    if (!rec) {
      return res.status(404).json({ error: 'Gasto recurrente no encontrado', id: req.params.id });
    }

    // 🔒 PENDIENTE = única vía permitida (se permite más de un abono mientras haya saldo)
    if (rec.estado !== 'PENDIENTE') {
      return res.status(400).json({ error: 'Solo se puede registrar el pago de una obligación PENDIENTE' });
    }

    const esVariable = !!rec.montoVariable;

    // 🧮 Monto REAL del periodo vigente:
    //  - FIJO: el monto del contrato (rec.monto), conocido desde el alta.
    //  - VARIABLE: el del recibo actual (rec.montoActual). Si el recibo aún no se
    //    ha capturado, el PRIMER abono del periodo lo fija usando montoRecibo; los
    //    abonos siguientes del mismo periodo usan ese mismo monto real.
    let montoReal;
    let reciboNuevo = null;
    if (esVariable) {
      if (rec.montoActual != null) {
        montoReal = round2(Number(rec.montoActual) || 0);
      } else {
        const recibo = round2(Number(montoRecibo));
        if (!recibo || recibo <= 0) {
          return res.status(422).json({
            error: 'montoRecibo es requerido para fijar el monto real del periodo variable',
          });
        }
        montoReal = recibo;
        reciboNuevo = recibo;
      }
    } else {
      montoReal = round2(Number(rec.monto) || 0);
    }

    // 🔒 El estado se deriva SIEMPRE del saldo real acumulado (fijo O variable):
    // el monto del periodo NO cambia, los abonos se acumulan y nunca se sobrepaga
    // (pendiente nunca queda negativo).
    const acumulado = Number(rec.montoPagado) || 0;
    const pendiente = round2(Math.max(0, montoReal - acumulado));
    if (pendiente <= 0.005) {
      return res.status(400).json({ error: 'Esta obligación ya está pagada por completo' });
    }
    if (pago > pendiente + 0.005) {
      return res.status(400).json({
        error: `El pago supera el pendiente del periodo (pendiente: ${round2(pendiente)})`,
      });
    }

    // 1. Crear UN gasto real en Finanzas por el importe abonado. Pertenece al
    //    periodo financiero que esté abierto en este momento.
    const periodo = await getPeriodoAbierto();
    const finanza = await Finanza.create({
      tipo: 'GASTO',
      fecha: fechaPagoISO,
      concepto: rec.nombre,
      monto: pago,
      metodoPago,
      categoria: CATEGORIA_GASTO_FIJO,
      nota: rec.nota || `Pago recurrente: ${rec.nombre}`,
      usuario: req.user._id,
      cierreFinancieroId: periodo._id,
    });

    // 2. Aplicar de forma atómica con reintento optimista:
    //    el filtro exige el montoPagado leído, así un abono concurrente NO se pierde
    //    ni se cuenta dos veces.
    for (let intento = 0; intento < 10; intento++) {
      const recN = intento === 0 ? rec : await GastoRecurrente.findById(req.params.id);
      if (!recN || recN.estado !== 'PENDIENTE') {
        await Finanza.findByIdAndDelete(finanza._id);
        return res.status(400).json({ error: 'Esta obligación ya no está pendiente de pago' });
      }

      // 🧮 Monto real del periodo con datos frescos (puede cambiar si un abono
      // concurrente capturó el recibo variable a la mitad del camino)
      let montoRealIter;
      if (esVariable && recN.montoActual != null) {
        montoRealIter = round2(Number(recN.montoActual));
      } else if (esVariable) {
        montoRealIter = reciboNuevo;
      } else {
        montoRealIter = round2(Number(recN.monto) || 0);
      }

      const acumulado = Number(recN.montoPagado) || 0;
      const pendiente = round2(Math.max(0, montoRealIter - acumulado));

      if (pago > pendiente + 0.005) {
        await Finanza.findByIdAndDelete(finanza._id);
        return res.status(400).json({
          error: `El pago supera el pendiente del periodo (pendiente: ${round2(pendiente)})`,
        });
      }

      // 🔁 Acumula el abono y deriva el estado del saldo resultante
      const nuevoAcumulado = round2(acumulado + pago);
      const nuevoPendiente = round2(Math.max(0, montoRealIter - nuevoAcumulado));
      const nuevoEstado = nuevoPendiente <= 0.005 ? 'PAGADO' : 'PENDIENTE';

      const actualizado = await GastoRecurrente.findOneAndUpdate(
        {
          _id: recN._id,
          estado: 'PENDIENTE',
          montoPagado: recN.montoPagado, // guarda optimista: debe coincidir con lo leído
          montoActual: recN.montoActual, // y el recibo variable no debe fijarse a mitad de camino
        },
        {
          $set: {
            estado: nuevoEstado,
            fechaPago: fechaPagoISO,
            montoPagado: nuevoAcumulado,
            metodoPago,
            finanzaId: finanza._id,
            // 💰 El primer abono del periodo variable fija su monto real
            ...(esVariable && recN.montoActual == null && reciboNuevo != null ? { montoActual: reciboNuevo } : {}),
          },
          $push: {
            // 📸 Snapshot completo del abono (inmune a renames y a edición/borrado del Finanza)
            historial: {
              fecha: fechaPagoISO,
              monto: pago,
              finanzaId: finanza._id,
              concepto: recN.nombre,
              metodoPago,
              nota: recN.nota || `Pago recurrente: ${recN.nombre}`,
              categoria: CATEGORIA_GASTO_FIJO,
            },
          },
        },
        { new: true }
      );

      if (actualizado) {
        return res.status(200).json({ recurrente: actualizado, finanza });
      }
      // Otro abono concurrente modificó el documento: reintentar con datos frescos
    }

    // 3. No se pudo aplicar tras varios reintentos → deshacer el gasto creado
    await Finanza.findByIdAndDelete(finanza._id);
    return res.status(409).json({
      error: 'No se pudo registrar el pago por una actualización concurrente, intenta de nuevo',
    });
  } catch (err) {
    if (err.name === 'ValidationError') {
      const details = Object.values(err.errors).map((e) => ({ field: e.path, message: e.message }));
      return res.status(422).json({ error: 'Validation failed', details });
    }
    res.status(500).json({ error: 'Error al registrar el pago' });
  }
};

// POST /api/gastos-recurrentes/:id/renovar
export const renovar = async (req, res) => {
  try {
    const rec = await GastoRecurrente.findById(req.params.id);
    if (!rec) {
      return res.status(404).json({ error: 'Gasto recurrente no encontrado', id: req.params.id });
    }

    // 🔒 Regla de oro: nunca avanzar con saldo pendiente. El estado correcto se deriva
    // del saldo acumulado, así la operación falla aunque algún dato previo diga PAGADO
    // y nunca se pierden abonos parciales.
    const esVariable = !!rec.montoVariable;
    const acumulado = Number(rec.montoPagado) || 0;
    const montoReal = esVariable ? rec.montoActual : Number(rec.monto) || 0;

    // 🌀 Para VARIABLE sin recibo capturado tampoco se avanza: NO se considera pagado
    if (esVariable && montoReal == null) {
      return res.status(400).json({
        error: 'No se puede avanzar al siguiente periodo: falta capturar el monto del recibo de este periodo (VARIABLE).',
      });
    }

    const pendiente = round2(Math.max(0, Number(montoReal) - acumulado));

    if (pendiente > 0.005) {
      return res.status(400).json({
        error: `No se puede avanzar al siguiente periodo: todavía existe un saldo pendiente de $${round2(pendiente)}.`,
      });
    }

    if (rec.estado !== 'PAGADO') {
      return res.status(400).json({ error: 'Primero registra el pago para usar "Siguiente periodo"' });
    }

    let nuevaFecha;
    if (rec.periodicidad === 'OTRA') {
      if (!req.body.nuevaFecha || !esFechaValida(req.body.nuevaFecha)) {
        return res.status(422).json({ error: 'nuevaFecha es requerida para periodicidad OTRA' });
      }
      nuevaFecha = fechaInicioDia(req.body.nuevaFecha);
    } else {
      nuevaFecha = siguienteVencimiento(rec.periodicidad, rec.proximoPago);
    }

    rec.estado = 'PENDIENTE';
    rec.proximoPago = nuevaFecha;
    rec.fechaPago = null;
    rec.montoPagado = null;
    rec.metodoPago = null;
    rec.finanzaId = null;
    if (esVariable) {
      // 💰 El nuevo periodo VARIABLE vuelve a quedar sin monto real: se captura al llegar el recibo.
      // El campo rec.monto (referencia/estimación) se conserva tal cual.
      rec.montoActual = null;
    }

    await rec.save();
    res.json(rec);
  } catch (err) {
    res.status(500).json({ error: 'Error al renovar gasto recurrente' });
  }
};

function validar(body) {
  const errores = [];

  if (!body.nombre || !String(body.nombre).trim()) {
    errores.push({ field: 'nombre', message: 'nombre es requerido' });
  }

  const montoVariable = !!body.montoVariable;
  if (!montoVariable) {
    // 🔒 FIJO: el monto es conocido y obligatorio desde el alta
    const monto = Number(body.monto);
    if (!monto || monto <= 0) {
      errores.push({ field: 'monto', message: 'monto debe ser un número mayor a 0 para gastos fijos' });
    }
  } else if (montoVariable && body.monto != null && body.monto !== '') {
    // VARIABLE: el monto es solo referencia/estimación (opcional)
    const monto = Number(body.monto);
    if (!Number.isFinite(monto) || monto <= 0) {
      errores.push({ field: 'monto', message: 'monto debe ser un número mayor a 0 si se indica' });
    }
  }

  if (!PERIODICIDADES.includes(body.periodicidad)) {
    errores.push({ field: 'periodicidad', message: 'periodicidad inválida' });
  }

  if (!body.proximoPago || !esFechaValida(body.proximoPago)) {
    errores.push({ field: 'proximoPago', message: 'proximoPago es requerido en formato YYYY-MM-DD' });
  }

  return errores;
}

    res.status(500).json({ error: 'Error al agregar deuda' });

  try {
    const { monto, descripcion } = req.body;
    const deudor = await ListaNegra.findById(req.params.id);

import Mesa from '../models/Mesa.js';

// 🔍 GET /api/mesas
export const getMesas = async (req, res) => {
  try {
    const mesas = await Mesa.find().sort({ numero: 1 });
    res.json(mesas.map(formatMesa));
  } catch (err) {
    res.status(500).json({ error: 'Error al obtener mesas' });
  }
};

// 🆕 POST /api/mesas
export const createMesa = async (req, res) => {
  try {

    const { nombre } = req.body;

    if (!nombre || !nombre.trim()) {
      return res.status(400).json({
        error: 'El nombre es obligatorio'
      });
    }

    const count = await Mesa.countDocuments();

    // 🔥 LÍMITE
    if (count >= 20) {
      return res.status(400).json({
        error: 'Límite de mesas alcanzado (20)',
      });
    }

    // 🔥 número automático
    const ultimaMesa = await Mesa.findOne().sort({ numero: -1 });
    const nuevoNumero = ultimaMesa ? ultimaMesa.numero + 1 : 1;

    const mesa = await Mesa.create({
      numero: nuevoNumero,
      nombre
    });

    res.status(201).json(formatMesa(mesa));

  } catch (err) {
    res.status(500).json({ error: 'Error al crear mesa' });
  }
};

// ✏️ PUT /api/mesas/:id
export const updateMesa = async (req, res) => {
  try {
    const mesa = await Mesa.findByIdAndUpdate(
      req.params.id,
      req.body,
      { new: true, runValidators: true }
    );

    if (!mesa) {
      return res.status(404).json({ error: 'Mesa no encontrada' });
    }

    res.json(formatMesa(mesa));

  } catch (err) {
    res.status(500).json({ error: 'Error al actualizar mesa' });
  }
};

// ❌ DELETE /api/mesas/:id
export const deleteMesa = async (req, res) => {
  try {
    const mesa = await Mesa.findById(req.params.id);

    if (!mesa) {
      return res.status(404).json({ error: 'Mesa no encontrada' });
    }

    // 🔥 no eliminar si está ocupada
    if (mesa.estado === 'OCUPADA') {
      return res.status(400).json({
        error: 'No puedes eliminar una mesa ocupada'
      });
    }

    await mesa.deleteOne();

    res.json({ message: 'Mesa eliminada', id: req.params.id });

  } catch (err) {
    res.status(500).json({ error: 'Error al eliminar mesa' });
  }
};

// 🔥 FORMAT
function formatMesa(doc) {
  return {
    id: doc._id.toString(),
    numero: doc.numero,
    nombre: doc.nombre,
    estado: doc.estado,
    createdAt: doc.createdAt,
    updatedAt: doc.updatedAt,
  };
}

import Empleado from '../models/Empleado.js';
import PagoNomina from '../models/PagoNomina.js';

const MS_DIA = 86400000;

const METODOS_PAGO = ['EFECTIVO', 'TARJETA', 'TRANSFERENCIA'];
const PERIODICIDADES = ['SEMANAL', 'QUINCENAL', 'MENSUAL', 'BIMESTRAL', 'TRIMESTRAL', 'SEMESTRAL', 'ANUAL', 'OTRA'];
// RECONOCE TODOS los tipos (incluye históricos EXTRAORDINARIO/ADELANTO para
// lectura/anulación), pero SOLO estos se aceptan para NUEVOS pagos:
const TIPOS_PAGO = ['PERIODICO', 'BONO', 'EXTRAORDINARIO', 'ADELANTO', 'LIQUIDACION'];
const TIPOS_PAGO_NUEVOS = ['PERIODICO', 'BONO', 'LIQUIDACION'];
const PERIODICIDADES_ANCLADAS = ['MENSUAL', 'BIMESTRAL', 'TRIMESTRAL', 'SEMESTRAL', 'ANUAL'];

function limpiarTexto(valor) {
  return typeof valor === 'string' ? valor.trim() : '';
}

function aNumero(valor) {
  const n = Number(valor);
  return Number.isFinite(n) ? n : NaN;
}

function mismoDia(a, b) {
  return a instanceof Date && b instanceof Date && a.getTime() === b.getTime();
}

function iguales(a, b) {
  if (a instanceof Date || b instanceof Date) return mismoDia(a, b);
  return a === b;
}

// 📅 Quincena de calendario real: 15 → 30 (o último día si el mes tiene menos)
function anclarQuincena(fecha) {
  const y = fecha.getUTCFullYear();
  const m = fecha.getUTCMonth();
  const dia = fecha.getUTCDate();

  if (dia <= 15) return new Date(Date.UTC(y, m, 15));
  return new Date(Date.UTC(y, m, Math.min(30, ultimoDiaDelMes(fecha))));
}

// 🎯 Normaliza la fecha de calendario según la periodicidad (reglas UTC)
function normalizarProximoPago(periodicidad, fecha, diaCorte = null) {
  if (!(fecha instanceof Date) || Number.isNaN(fecha.getTime())) return null;

  const base = new Date(fecha.getTime());
  base.setUTCHours(0, 0, 0, 0);

  if (periodicidad === 'QUINCENAL') return anclarQuincena(base);
  if (PERIODICIDADES_ANCLADAS.includes(periodicidad) && diaCorte) {
    const ultimo = ultimoDiaDelMes(base);
    return new Date(Date.UTC(base.getUTCFullYear(), base.getUTCMonth(), Math.min(diaCorte, ultimo)));
  }
  return base;
}

// 🧩 Agrega los campos calculados (estado visual y estado del periodo) sin persistir
function conEstado(empleado, opciones = {}) {
  const hoy = opciones.hoy || hoyUTC();
  const visual =
    empleado.estado === 'ACTIVO'
      ? estadoPagoNomina(empleado.proximoPago, hoy)
      : { estadoVisual: 'INACTIVO', diasRestantes: null };

  // 💰 Estado del periodo: comparar lo acumulado contra lo esperado (sueldo)
  const esperado = Number(empleado.sueldo || 0);
  const acumuladoPeriodo = Math.min(Number(opciones.acumuladoPeriodo || 0), esperado);
  const pendientePeriodo = Math.max(0, esperado - acumuladoPeriodo);
  const pagadoPeriodoActual = esperado > 0 && acumuladoPeriodo >= esperado;

  return {
    ...empleado,
    proximoPagoISO: aFechaISO(empleado.proximoPago),
    estadoVisual: visual.estadoVisual,
    diasRestantes: visual.diasRestantes,
    pagadoPeriodoActual,
    acumuladoPeriodo,
    pendientePeriodo,
    puedePagar: empleado.estado === 'ACTIVO' && !pagadoPeriodoActual,
  };
}

// 💰 Acumulado de pagos periódicos vigentes POR EMPLEADO y POR PERIODO.
// Devuelve Map<empleadoId, Map<periodoMs, acumulado>>.
async function periodosPagados(empleados) {
  const ids = empleados.map((e) => e._id);
  if (ids.length === 0) return new Map();

  const agrupados = await PagoNomina.aggregate([
    { $match: { empleado: { $in: ids }, tipoPago: 'PERIODICO', anulado: false } },
    { $group: { _id: { empleado: '$empleado', periodo: '$periodo' }, monto: { $sum: '$monto' } } },
  ]);

  const mapa = new Map();
  for (const g of agrupados) {
    const empleadoId = String(g._id.empleado);
    if (!mapa.has(empleadoId)) mapa.set(empleadoId, new Map());
    mapa.get(empleadoId).set(g._id.periodo.getTime(), g.monto);
  }
  return mapa;
}

// ♻️ Recalcula los campos de "último pago" con los pagos que siguen vigentes
async function sincronizarUltimoPago(empleadoId) {
  const ultimo = await PagoNomina.findOne({ empleado: empleadoId, anulado: false })
    .sort({ fechaPago: -1, createdAt: -1 })
    .lean();

  if (!ultimo) {
    await Empleado.updateOne(
      { _id: empleadoId },
      { $set: { ultimoPagoId: null, fechaUltimoPago: null, montoUltimoPago: null, metodoPagoUltimo: null } }
    );
    return;
  }

  await Empleado.updateOne(
    { _id: empleadoId },
    {
      $set: {
        ultimoPagoId: ultimo._id,
        fechaUltimoPago: ultimo.fechaPago,
        montoUltimoPago: ultimo.monto,
        metodoPagoUltimo: ultimo.metodoPago,
      },
    }
  );
}

// 🧾 Valida el cuerpo de alta/edición de empleado
function validarEmpleado(body, { parcial = false } = {}) {
  const errores = [];

  const requiere = (campo) => !parcial || body[campo] !== undefined;

  if (requiere('nombre')) {
    const nombre = limpiarTexto(body.nombre);
    if (!nombre) errores.push({ field: 'nombre', message: 'nombre es requerido' });
    else if (nombre.length > 100) errores.push({ field: 'nombre', message: 'nombre debe tener 100 caracteres o menos' });
  }

  if (requiere('puesto')) {
    const puesto = limpiarTexto(body.puesto);
    if (!puesto) errores.push({ field: 'puesto', message: 'puesto es requerido' });
    else if (puesto.length > 80) errores.push({ field: 'puesto', message: 'puesto debe tener 80 caracteres o menos' });
  }

  if (requiere('sueldo')) {
    const sueldo = aNumero(body.sueldo);
    if (!Number.isFinite(sueldo) || sueldo <= 0) {
      errores.push({ field: 'sueldo', message: 'sueldo debe ser un número mayor a 0' });
    }
  }

  if (requiere('periodicidad') && !PERIODICIDADES.includes(body.periodicidad)) {
    errores.push({ field: 'periodicidad', message: 'periodicidad es requerida' });
  }

  if (requiere('proximoPago') && body.proximoPago !== null && body.proximoPago !== undefined) {
    if (!esFechaValida(body.proximoPago)) {
      errores.push({ field: 'proximoPago', message: 'proximoPago es requerido en formato YYYY-MM-DD' });
    }
  }

  if (body.diaCorte !== undefined && body.diaCorte !== null && body.diaCorte !== '') {
    const dia = aNumero(body.diaCorte);
    if (!Number.isInteger(dia) || dia < 1 || dia > 31) {
      errores.push({ field: 'diaCorte', message: 'diaCorte debe ser un número entero entre 1 y 31' });
    }
  }

  if (body.diasIntervalo !== undefined && body.diasIntervalo !== null && body.diasIntervalo !== '') {
    const dias = aNumero(body.diasIntervalo);
    if (!Number.isInteger(dias) || dias < 1 || dias > 365) {
      errores.push({ field: 'diasIntervalo', message: 'diasIntervalo debe ser un número entero entre 1 y 365' });
    }
  }

  const periodicidad = body.periodicidad;

  if (periodicidad === 'OTRA') {
    const dias = aNumero(body.diasIntervalo);
    if (!parcial && (!Number.isInteger(dias) || dias < 1 || dias > 365)) {
      errores.push({ field: 'diasIntervalo', message: 'periodicidad OTRA requiere diasIntervalo entre 1 y 365' });
    }
  }

  if (periodicidad && PERIODICIDADES_ANCLADAS.includes(periodicidad)) {
    const dia = aNumero(body.diaCorte);
    if (!parcial && (!Number.isInteger(dia) || dia < 1 || dia > 31)) {
      errores.push({ field: 'diaCorte', message: `periodicidad ${periodicidad} requiere diaCorte entre 1 y 31` });
    }
  }

  if (body.fechaIngreso !== undefined && body.fechaIngreso !== null && body.fechaIngreso !== '') {
    if (!esFechaValida(body.fechaIngreso)) {
      errores.push({ field: 'fechaIngreso', message: 'fechaIngreso debe estar en formato YYYY-MM-DD' });
    }
  }

  if (body.notas !== undefined) {
    const notas = limpiarTexto(body.notas);
    if (notas.length > 300) errores.push({ field: 'notas', message: 'notas debe tener 300 caracteres o menos' });
  }

  if (body.estado !== undefined && !['ACTIVO', 'INACTIVO'].includes(body.estado)) {
    errores.push({ field: 'estado', message: 'estado debe ser ACTIVO o INACTIVO' });
  }

  return errores;
}

// GET /api/nomina/resumen
export const getResumen = async (req, res) => {
  try {
    const hoy = hoyUTC();
    const empleados = await Empleado.find().lean();
    const pagados = await periodosPagados(empleados);
    const proyectados = empleados.map((e) => {
      const porPeriodo = pagados.get(String(e._id));
      const acumulado = porPeriodo?.get(new Date(e.proximoPago).getTime()) || 0;
      return conEstado(e, { hoy, acumuladoPeriodo: acumulado });
    });
    const activos = proyectados.filter((e) => e.estado === 'ACTIVO');

    const pendientes = activos.filter((e) => e.estadoVisual === 'PENDIENTE');
    const proximos = activos.filter((e) => e.estadoVisual === 'PROXIMO');

    const inicioMes = new Date(Date.UTC(hoy.getUTCFullYear(), hoy.getUTCMonth(), 1));
    const pagosMes = await PagoNomina.find({
      anulado: false,
      fechaPago: { $gte: inicioMes, $lte: hoy },
    })
      .select('monto')
      .lean();

    const pagosMesTotal = pagosMes.reduce((suma, p) => suma + p.monto, 0);

    const cola = [...pendientes, ...proximos].sort((a, b) => {
      if (a.diasRestantes !== b.diasRestantes) return a.diasRestantes - b.diasRestantes;
      return a.nombre.localeCompare(b.nombre);
    });

    res.json({
      fecha: aFechaISO(hoy),
      empleados: {
        activos: activos.length,
        inactivos: empleados.length - activos.length,
        total: empleados.length,
      },
      pendientes: {
        cantidad: pendientes.length,
        monto: pendientes.reduce((suma, e) => suma + e.pendientePeriodo, 0),
      },
      proximos: {
        cantidad: proximos.length,
        monto: proximos.reduce((suma, e) => suma + e.pendientePeriodo, 0),
      },
      mes: {
        cantidad: pagosMes.length,
        monto: pagosMesTotal,
      },
      proximosPagos: cola.slice(0, 5).map((e) => ({
        empleadoId: e._id,
        nombre: e.nombre,
        puesto: e.puesto,
        monto: e.sueldo,
        proximoPago: e.proximoPagoISO,
        diasRestantes: e.diasRestantes,
        estadoVisual: e.estadoVisual,
      })),
    });
  } catch (err) {
    console.error('Error en getResumen (nomina):', err);
    res.status(500).json({ error: 'Error al obtener el resumen de nómina' });
  }
};

// GET /api/nomina/empleados
export const getEmpleados = async (req, res) => {
  try {
    const hoy = hoyUTC();
    const filtro = {};

    if (req.query.estado && ['ACTIVO', 'INACTIVO'].includes(req.query.estado)) {
      filtro.estado = req.query.estado;
    }

    if (limpiarTexto(req.query.q)) {
      const q = limpiarTexto(req.query.q).replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
      filtro.$or = [{ nombre: { $regex: q, $options: 'i' } }, { puesto: { $regex: q, $options: 'i' } }];
    }

    const empleados = await Empleado.find(filtro)
      .populate('usuario', 'name username')
      .sort({ estado: 1, proximoPago: 1, nombre: 1 })
      .lean();

    const pagados = await periodosPagados(empleados);

    const resultado = empleados.map((empleado) => {
      const porPeriodo = pagados.get(String(empleado._id));
      const acumulado = porPeriodo?.get(new Date(empleado.proximoPago).getTime()) || 0;
      return conEstado(empleado, { hoy, acumuladoPeriodo: acumulado });
    });

    res.json(resultado);
  } catch (err) {
    console.error('Error en getEmpleados (nomina):', err);
    res.status(500).json({ error: 'Error al obtener los empleados' });
  }
};

// GET /api/nomina/empleados/:id
export const getEmpleado = async (req, res) => {
  try {
    const hoy = hoyUTC();
    const empleado = await Empleado.findById(req.params.id)
      .populate('usuario', 'name username')
      .lean();

    if (!empleado) {
      return res.status(404).json({ error: 'Empleado no encontrado', id: req.params.id });
    }

    const pagos = await PagoNomina.find({ empleado: empleado._id })
      .populate('usuario', 'name username')
      .populate('anuladoPor', 'name username')
      .sort({ fechaPago: -1, createdAt: -1 })
      .limit(50)
      .lean();

    const pagados = await periodosPagados([empleado]);
    const porPeriodo = pagados.get(String(empleado._id));
    const acumulado = porPeriodo?.get(new Date(empleado.proximoPago).getTime()) || 0;
    const proyectado = conEstado(empleado, { hoy, acumuladoPeriodo: acumulado });

    res.json({ ...proyectado, pagos });
  } catch (err) {
    if (err.name === 'CastError') {
      return res.status(400).json({ error: 'ID de empleado inválido' });
    }
    console.error('Error en getEmpleado (nomina):', err);
    res.status(500).json({ error: 'Error al obtener el empleado' });
  }
};

// POST /api/nomina/empleados
export const crearEmpleado = async (req, res) => {
  try {
    const errores = validarEmpleado(req.body);
    if (errores.length > 0) {
      return res.status(422).json({ error: 'Validation failed', details: errores });
    }

    const periodicidad = req.body.periodicidad;
    const diaCorte = aNumero(req.body.diaCorte) || null;
    const diasIntervalo = aNumero(req.body.diasIntervalo) || null;
    const fechaIngreso = req.body.fechaIngreso ? fechaInicioDia(req.body.fechaIngreso) : null;

    const proximoPago = normalizarProximoPago(
      periodicidad,
      req.body.proximoPago ? fechaInicioDia(req.body.proximoPago) : hoyUTC(),
      diaCorte
    );

    const nuevo = await Empleado.create({
      nombre: limpiarTexto(req.body.nombre),
      puesto: limpiarTexto(req.body.puesto),
      sueldo: aNumero(req.body.sueldo),
      periodicidad,
      diaCorte: PERIODICIDADES_ANCLADAS.includes(periodicidad) ? diaCorte : null,
      diasIntervalo: periodicidad === 'OTRA' ? diasIntervalo : null,
      proximoPago,
      estado: req.body.estado === 'INACTIVO' ? 'INACTIVO' : 'ACTIVO',
      fechaIngreso,
      notas: limpiarTexto(req.body.notas),
      usuario: req.user._id,
      historialCambios: [
        {
          fecha: new Date(),
          campo: 'alta',
          antes: null,
          despues: `${limpiarTexto(req.body.nombre)} · ${periodicidad} · $${aNumero(req.body.sueldo)}`,
          usuario: req.user._id,
        },
      ],
    });

    res.status(201).json(conEstado(nuevo.toObject(), { acumuladoPeriodo: 0 }));
  } catch (err) {
    if (err.name === 'ValidationError') {
      const details = Object.values(err.errors).map((e) => ({ field: e.path, message: e.message }));
      return res.status(422).json({ error: 'Validation failed', details });
    }
    console.error('Error en crearEmpleado (nomina):', err);
    res.status(500).json({ error: 'Error al crear el empleado' });
  }
};

// PUT /api/nomina/empleados/:id
export const actualizarEmpleado = async (req, res) => {
  try {
    const empleado = await Empleado.findById(req.params.id);
    if (!empleado) {
      return res.status(404).json({ error: 'Empleado no encontrado', id: req.params.id });
    }

    const errores = validarEmpleado(req.body, { parcial: true });
    if (errores.length > 0) {
      return res.status(422).json({ error: 'Validation failed', details: errores });
    }

    const cambios = [];
    const registrar = (campo, antes, despues) => {
      if (!iguales(antes, despues)) {
        cambios.push({ fecha: new Date(), campo, antes, despues, usuario: req.user._id });
      }
    };

    if (req.body.nombre !== undefined) {
      const nombre = limpiarTexto(req.body.nombre);
      registrar('nombre', empleado.nombre, nombre);
      empleado.nombre = nombre;
    }

    if (req.body.puesto !== undefined) {
      const puesto = limpiarTexto(req.body.puesto);
      registrar('puesto', empleado.puesto, puesto);
      empleado.puesto = puesto;
    }

    if (req.body.sueldo !== undefined) {
      const sueldo = aNumero(req.body.sueldo);
      registrar('sueldo', empleado.sueldo, sueldo);
      empleado.sueldo = sueldo;
    }

    if (req.body.periodicidad !== undefined) {
      registrar('periodicidad', empleado.periodicidad, req.body.periodicidad);
      empleado.periodicidad = req.body.periodicidad;
    }

    if (req.body.diaCorte !== undefined) {
      const diaCorte = req.body.diaCorte === null || req.body.diaCorte === '' ? null : aNumero(req.body.diaCorte);
      registrar('diaCorte', empleado.diaCorte, diaCorte);
      empleado.diaCorte = diaCorte;
    }

    if (req.body.diasIntervalo !== undefined) {
      const dias = req.body.diasIntervalo === null || req.body.diasIntervalo === '' ? null : aNumero(req.body.diasIntervalo);
      registrar('diasIntervalo', empleado.diasIntervalo, dias);
      empleado.diasIntervalo = dias;
    }

    if (req.body.fechaIngreso !== undefined) {
      const fechaIngreso = req.body.fechaIngreso ? fechaInicioDia(req.body.fechaIngreso) : null;
      registrar('fechaIngreso', empleado.fechaIngreso, fechaIngreso);
      empleado.fechaIngreso = fechaIngreso;
    }

    if (req.body.notas !== undefined) {
      const notas = limpiarTexto(req.body.notas);
      registrar('notas', empleado.notas, notas);
      empleado.notas = notas;
    }

    if (req.body.estado !== undefined) {
      registrar('estado', empleado.estado, req.body.estado);
      empleado.estado = req.body.estado;
    }

    // 📅 El calendario solo se mueve si viene explícito en la edición
    if (req.body.proximoPago !== undefined && req.body.proximoPago) {
      const fecha = fechaInicioDia(req.body.proximoPago);
      const normalizado = normalizarProximoPago(empleado.periodicidad, fecha, empleado.diaCorte);
      registrar('proximoPago', empleado.proximoPago, normalizado);
      empleado.proximoPago = normalizado;
    }

    // 🗂️ El historial guarda campo, valor anterior, valor nuevo, fecha y usuario.
    // Los pagos ya registrados NUNCA se tocan (conservan su snapshot).
    if (cambios.length > 0) {
      empleado.historialCambios.push(...cambios);
    }

    await empleado.save();

    const pagados = await periodosPagados([empleado.toObject()]);
    const porPeriodo = pagados.get(String(empleado._id));
    const acumulado = porPeriodo?.get(new Date(empleado.proximoPago).getTime()) || 0;

    res.json(conEstado(empleado.toObject(), { acumuladoPeriodo: acumulado }));
  } catch (err) {
    if (err.name === 'CastError') {
      return res.status(400).json({ error: 'ID de empleado inválido' });
    }
    if (err.name === 'ValidationError') {
      const details = Object.values(err.errors).map((e) => ({ field: e.path, message: e.message }));
      return res.status(422).json({ error: 'Validation failed', details });
    }
    console.error('Error en actualizarEmpleado (nomina):', err);
    res.status(500).json({ error: 'Error al actualizar el empleado' });
  }
};

// DELETE /api/nomina/empleados/:id
export const eliminarEmpleado = async (req, res) => {
  try {
    const empleado = await Empleado.findById(req.params.id);
    if (!empleado) {
      return res.status(404).json({ error: 'Empleado no encontrado', id: req.params.id });
    }

    const pagos = await PagoNomina.countDocuments({ empleado: empleado._id });
    if (pagos > 0) {
      return res.status(400).json({
        error: 'Este empleado tiene pagos registrados. Márcalo como INACTIVO para conservar el historial.',
      });
    }

    await empleado.deleteOne();
    res.json({ message: 'Empleado eliminado', id: req.params.id });
  } catch (err) {
    if (err.name === 'CastError') {
      return res.status(400).json({ error: 'ID de empleado inválido' });
    }
    console.error('Error en eliminarEmpleado (nomina):', err);
    res.status(500).json({ error: 'Error al eliminar el empleado' });
  }
};

// POST /api/nomina/empleados/:id/renovar
export const renovarCalendario = async (req, res) => {
  try {
    const empleado = await Empleado.findById(req.params.id);
    if (!empleado) {
      return res.status(404).json({ error: 'Empleado no encontrado', id: req.params.id });
    }

    if (empleado.estado !== 'ACTIVO') {
      return res.status(400).json({ error: 'No se puede renovar el calendario de un empleado INACTIVO' });
    }

    let nuevoProximoPago;

    if (req.body.nuevaFecha) {
      if (!esFechaValida(req.body.nuevaFecha)) {
        return res.status(422).json({ error: 'Validation failed', details: [{ field: 'nuevaFecha', message: 'nuevaFecha debe estar en formato YYYY-MM-DD' }] });
      }
      nuevoProximoPago = normalizarProximoPago(
        empleado.periodicidad,
        fechaInicioDia(req.body.nuevaFecha),
        empleado.diaCorte
      );
    } else {
      nuevoProximoPago = siguienteVencimientoNomina({
        periodicidad: empleado.periodicidad,
        base: empleado.proximoPago,
        diaCorte: empleado.diaCorte,
        diasIntervalo: empleado.diasIntervalo,
      });
    }

    if (!nuevoProximoPago || !Number.isFinite(nuevoProximoPago.getTime())) {
      return res.status(422).json({ error: 'No fue posible calcular el siguiente pago' });
    }

    const anterior = empleado.proximoPago;
    empleado.proximoPago = nuevoProximoPago;
    empleado.historialCambios.push({
      fecha: new Date(),
      campo: 'proximoPago',
      antes: anterior,
      despues: nuevoProximoPago,
      usuario: req.user._id,
    });

    await empleado.save();

    res.json(conEstado(empleado.toObject(), { acumuladoPeriodo: 0 }));
  } catch (err) {
    if (err.name === 'CastError') {
      return res.status(400).json({ error: 'ID de empleado inválido' });
    }
    console.error('Error en renovarCalendario (nomina):', err);
    res.status(500).json({ error: 'Error al renovar el calendario de pago' });
  }
};

// GET /api/nomina/pagos
export const getPagos = async (req, res) => {
  try {
    const filtro = {};

    if (req.query.empleado) {
      const existe = await Empleado.exists({ _id: req.query.empleado });
      if (!existe) {
        return res.status(400).json({ error: 'ID de empleado inválido' });
      }
      filtro.empleado = req.query.empleado;
    }

    // 📅 Modo "mes calendario": year + month juntos. El mes es el calendario REAL
    // del pago (fechaPago), no el periodo que cubre. Sin cierre mensual manual.
    const anioQuery = req.query.year !== undefined ? Number(req.query.year) : undefined;
    const mesQuery = req.query.month !== undefined ? Number(req.query.month) : undefined;
    const esMes = anioQuery !== undefined || mesQuery !== undefined;

    if (esMes && (anioQuery === undefined || mesQuery === undefined)) {
      return res.status(400).json({ error: 'Los filtros "year" y "month" deben enviarse juntos' });
    }

    if (esMes) {
      if (!Number.isInteger(anioQuery) || anioQuery < 2000 || anioQuery > 2100) {
        return res.status(400).json({ error: 'El filtro "year" es inválido' });
      }
      if (!Number.isInteger(mesQuery) || mesQuery < 1 || mesQuery > 12) {
        return res.status(400).json({ error: 'El filtro "month" debe ser un número entre 1 y 12' });
      }
      filtro.fechaPago = {
        $gte: new Date(Date.UTC(anioQuery, mesQuery - 1, 1)),
        $lt: new Date(Date.UTC(anioQuery, mesQuery, 1)),
      };
    }

    if (!esMes && req.query.desde) {
      if (!esFechaValida(req.query.desde)) {
        return res.status(400).json({ error: 'El filtro "desde" debe estar en formato YYYY-MM-DD' });
      }
      filtro.fechaPago = { ...(filtro.fechaPago || {}), $gte: fechaInicioDia(req.query.desde) };
    }

    if (!esMes && req.query.hasta) {
      if (!esFechaValida(req.query.hasta)) {
        return res.status(400).json({ error: 'El filtro "hasta" debe estar en formato YYYY-MM-DD' });
      }
      const fin = addDays(fechaInicioDia(req.query.hasta), 1);
      filtro.fechaPago = { ...(filtro.fechaPago || {}), $lt: fin };
    }

    const incluirAnulados = String(req.query.incluirAnulados ?? 'true') !== 'false';
    if (!incluirAnulados) filtro.anulado = false;

    const limite = esMes ? 0 : Math.min(Math.max(Number(req.query.limite) || 100, 1), 500);

    const query = PagoNomina.find(filtro)
      .populate('usuario', 'name username')
      .populate('anuladoPor', 'name username')
      .sort({ fechaPago: -1, createdAt: -1 });

    if (limite > 0) query.limit(limite);

    const pagos = await query.lean();

    // 🧮 Totales EXACTOS del conjunto filtrado (mes completo si es modo mes),
    // no del slice: los pagos anulados NUNCA suman al total vigente.
    const totales = await PagoNomina.aggregate([
      { $match: filtro },
      { $group: { _id: '$anulado', cantidad: { $sum: 1 }, monto: { $sum: '$monto' } } },
    ]);

    const vigente = { cantidad: 0, monto: 0 };
    const anulados = { cantidad: 0, monto: 0 };
    for (const t of totales) {
      const destino = t._id ? anulados : vigente;
      destino.cantidad += t.cantidad;
      destino.monto += t.monto;
    }

    res.json({ pagos, total: pagos.length, vigente, anulados });
  } catch (err) {
    if (err.name === 'CastError') {
      return res.status(400).json({ error: 'Filtro inválido' });
    }
    console.error('Error en getPagos (nomina):', err);
    res.status(500).json({ error: 'Error al obtener el historial de pagos' });
  }
};

// POST /api/nomina/pagos/registrar
export const registrarPago = async (req, res) => {
  try {
    const body = req.body || {};
    const errores = [];

    if (!body.empleado) {
      errores.push({ field: 'empleado', message: 'empleado es requerido' });
    }
    if (!METODOS_PAGO.includes(body.metodoPago)) {
      errores.push({ field: 'metodoPago', message: 'metodoPago es requerido (EFECTIVO, TARJETA o TRANSFERENCIA)' });
    }

    let tipoPago = body.tipoPago;
    if (tipoPago === undefined || tipoPago === null || tipoPago === '') {
      tipoPago = 'PERIODICO';
    }
    if (!TIPOS_PAGO.includes(tipoPago)) {
      errores.push({ field: 'tipoPago', message: 'tipoPago inválido' });
    } else if (!TIPOS_PAGO_NUEVOS.includes(tipoPago)) {
      errores.push({
        field: 'tipoPago',
        message: 'EXTRAORDINARIO y ADELANTO ya no están disponibles para nuevos pagos. Usa BONO o LIQUIDACION.',
      });
    }

    if (body.fechaPago && !esFechaValida(body.fechaPago)) {
      errores.push({ field: 'fechaPago', message: 'fechaPago debe estar en formato YYYY-MM-DD' });
    }

    if (body.periodo && !esFechaValida(body.periodo)) {
      errores.push({ field: 'periodo', message: 'periodo debe estar en formato YYYY-MM-DD' });
    }

    if (tipoPago !== 'PERIODICO' && !limpiarTexto(body.concepto)) {
      errores.push({ field: 'concepto', message: `concepto es requerido para pagos ${tipoPago}` });
    }

    if (body.nota && limpiarTexto(body.nota).length > 300) {
      errores.push({ field: 'nota', message: 'nota debe tener 300 caracteres o menos' });
    }

    const empleado = body.empleado ? await Empleado.findById(body.empleado) : null;
    if (body.empleado && !empleado) {
      return res.status(404).json({ error: 'Empleado no encontrado', id: body.empleado });
    }

    if (errores.length > 0) {
      return res.status(422).json({ error: 'Validation failed', details: errores });
    }

    if (empleado.estado !== 'ACTIVO') {
      return res.status(400).json({ error: 'No se pueden registrar pagos a empleados INACTIVOS' });
    }

    const monto = body.monto === undefined || body.monto === null ? empleado.sueldo : aNumero(body.monto);
    if (!Number.isFinite(monto) || monto <= 0) {
      return res.status(422).json({
        error: 'Validation failed',
        details: [{ field: 'monto', message: 'monto debe ser un número mayor a 0' }],
      });
    }

    const hoy = hoyUTC();
    const fechaPago = body.fechaPago ? fechaInicioDia(body.fechaPago) : hoy;

    if (fechaPago.getTime() > hoy.getTime()) {
      return res.status(422).json({
        error: 'Validation failed',
        details: [{ field: 'fechaPago', message: 'fechaPago no puede ser futura' }],
      });
    }

    // 🔒 Los pagos diferentes al periódico NUNCA mueven el calendario del empleado
    let periodo = null;
    let siguienteProximoPago = null;

    if (tipoPago === 'PERIODICO') {
      periodo = body.periodo ? fechaInicioDia(body.periodo) : new Date(empleado.proximoPago.getTime());

      // 💰 El monto esperado del periodo es el sueldo completo; el abono se compara
      // contra lo que falta (acumulado de pagos periódicos vigentes del periodo).
      const esperado = Number(empleado.sueldo || 0);

      const totales = await PagoNomina.aggregate([
        {
          $match: {
            empleado: empleado._id,
            periodo,
            tipoPago: 'PERIODICO',
            anulado: false,
          },
        },
        { $group: { _id: null, monto: { $sum: '$monto' } } },
      ]);

      const acumulado = totales.length > 0 ? totales[0].monto : 0;
      const pendiente = Math.max(0, esperado - acumulado);

      // 🙅 Protección de duplicados: bloquea el MISMO abono exacto (mismo día y
      // monto) para el periodo, pero permite otros abonos parciales distintos.
      const duplicado = await PagoNomina.findOne({
        empleado: empleado._id,
        periodo,
        fechaPago,
        monto,
        tipoPago: 'PERIODICO',
        anulado: false,
      }).lean();

      if (duplicado) {
        return res.status(409).json({
          error: `Ya existe un pago idéntico de $${monto} para el periodo ${aFechaISO(periodo)} de este empleado`,
        });
      }

      // ‼️ No se permite sobrepagar: el abono no puede exceder lo que falta del periodo
      if (monto > pendiente) {
        return res.status(422).json({
          error: 'Validation failed',
          details: [
            {
              field: 'monto',
              message: `El abono excede lo pendiente del periodo (acumulado $${acumulado} de $${esperado}; faltan $${pendiente})`,
            },
          ],
        });
      }

      // 📅 El siguiente vencimiento SOLO se calcula cuando el periodo queda completo
      if (acumulado + monto >= esperado) {
        siguienteProximoPago = siguienteVencimientoNomina({
          periodicidad: empleado.periodicidad,
          base: periodo,
          diaCorte: empleado.diaCorte,
          diasIntervalo: empleado.diasIntervalo,
        });
      }
    }

    const concepto = limpiarTexto(body.concepto) || `Nómina de ${empleado.nombre}`;

    // 📸 Snapshot inmutable: los pagos históricos nunca cambian aunque se edite el empleado
    const pago = await PagoNomina.create({
      empleado: empleado._id,
      periodo: periodo || fechaPago,
      fechaPago,
      monto,
      metodoPago: body.metodoPago,
      tipoPago,
      concepto,
      nota: limpiarTexto(body.nota),
      snapshot: {
        nombre: empleado.nombre,
        puesto: empleado.puesto,
        montoBase: empleado.sueldo,
        periodicidad: empleado.periodicidad,
        fechaIngreso: empleado.fechaIngreso,
      },
      anulado: false,
      usuario: req.user._id,
    });

    // 📅 El calendario solo avanza con pagos periódicos y nunca hacia atrás
    if (tipoPago === 'PERIODICO' && siguienteProximoPago) {
      const actualizaciones = {
        ultimoPagoId: pago._id,
        fechaUltimoPago: pago.fechaPago,
        montoUltimoPago: pago.monto,
        metodoPagoUltimo: pago.metodoPago,
      };

      if (siguienteProximoPago.getTime() > empleado.proximoPago.getTime()) {
        actualizaciones.proximoPago = siguienteProximoPago;
      }

      await Empleado.updateOne({ _id: empleado._id }, { $set: actualizaciones });
    }

    res.status(201).json(pago);
  } catch (err) {
    if (err.code === 11000) {
      return res.status(409).json({ error: 'Ya existe un pago idéntico para ese empleado, periodo, fecha y monto' });
    }
    if (err.name === 'CastError') {
      return res.status(400).json({ error: 'ID de empleado inválido' });
    }
    if (err.name === 'ValidationError') {
      const details = Object.values(err.errors).map((e) => ({ field: e.path, message: e.message }));
      return res.status(422).json({ error: 'Validation failed', details });
    }
    console.error('Error en registrarPago (nomina):', err);
    res.status(500).json({ error: 'Error al registrar el pago' });
  }
};

    pago.anuladoAt = new Date();

    await pago.save();

    const empleado = await Empleado.findById(pago.empleado);

    if (empleado) {

          {
            $match: {
              empleado: empleado._id,
              periodo: pago.periodo,
              tipoPago: 'PERIODICO',
              anulado: false,
            },
          },
          { $group: { _id: null, monto: { $sum: '$monto' } } },
        ]);

        const acumulado = totales.length > 0 ? totales[0].monto : 0;

        // 🔄 Si el periodo quedó incompleto, el calendario vuelve a ese periodo

        if (acumulado < Number(empleado.sueldo || 0) && empleado.proximoPago.getTime() >= pago.periodo.getTime()) {

          await empleado.save();
        }

    }

    const actualizado = await PagoNomina.findById(pago._id)
      .populate('usuario', 'name username')
      .populate('anuladoPor', 'name username')
      .lean();

    res.json(actualizado);
  } catch (err) {
    if (err.name === 'CastError') {
      return res.status(400).json({ error: 'ID de pago inválido' });
    }
    console.error('Error en anularPago (nomina):', err);
    res.status(500).json({ error: 'Error al anular el pago' });
  }
};

});

    // 4. Ver si ya está en la orden

    const item = orden.productos.find(

    );

    if (!item) {
      return res.status(404).json({ error: 'Producto no está en la orden' });
    }

    const producto = await Product.findById(productId).populate('components.productId');
    if (!producto) {
      return res.status(404).json({ error: 'Producto no encontrado' });
    }

    if (item.cantidad > 1) {
      item.cantidad -= 1;
    } else {
      orden.productos = orden.productos.filter(
        p => p.productId.toString() !== productId
      );
    }

    // recalcular total
    orden.total = orden.productos.reduce(
      (sum, p) => sum + p.precio * p.cantidad,
      0
    );

    await updateStock(producto, 1);
    await orden.save();

    res.json(formatOrden(orden));

  } catch (err) {
    res.status(500).json({ error: 'Error al quitar producto' });
  }

async function validateStock(producto) {
  if ((producto.type || 'SIMPLE') === 'SIMPLE') {
    return producto.stock <= 0 ? 'Stock insuficiente' : null;
  }

  await producto.populate('components.productId');

  if (!producto.components?.length) {
    return 'La promoción no tiene productos configurados';
  }

  const insufficient = producto.components.find(
    (component) => !component.productId || component.productId.stock < component.quantity
  );

  return insufficient ? `Stock insuficiente para ${insufficient.productId?.name || 'un producto de la promoción'}` : null;
}

async function updateStock(producto, direction) {
  if ((producto.type || 'SIMPLE') === 'SIMPLE') {
    producto.stock += direction;
    await producto.save();
    return;
  }

  await producto.populate('components.productId');

  for (const component of producto.components) {
    component.productId.stock += direction * component.quantity;
    await component.productId.save();
  }
}

export const createProduct = async (req, res) => {

    const errors = [];
    if (!name) errors.push({ field: 'name', message: 'name is required' });

    const cleanComponents = normalizeComponents(components);
    if (type === 'SIMPLE' && (stock === undefined || stock < 0)) {
      errors.push({ field: 'stock', message: 'stock must be 0 or more' });
    }
    if (type === 'COMBO' && cleanComponents.length === 0) {
      errors.push({ field: 'components', message: 'combo must include at least one product' });
    }

    if (errors.length > 0) {
      return res.status(422).json({ error: 'Validation failed', details: errors });
    }

    if (type === 'COMBO') {
      const componentCount = await Product.countDocuments({
        _id: { $in: cleanComponents.map((component) => component.productId) },
      });

      if (componentCount !== cleanComponents.length) {
        return res.status(422).json({
          error: 'Validation failed',
          details: [{ field: 'components', message: 'one or more combo products do not exist' }],
        });
      }
    }

    const product = await Product.create({
      name,
      price,
      stock: type === 'COMBO' ? 0 : stock,
      category,
      type,
      components: type === 'COMBO' ? cleanComponents : [],
    });
    await product.populate('components.productId');
    res.status(201).json(formatProduct(product));
  } catch (err) {
    if (err.name === 'ValidationError') {
      const details = Object.values(err.errors).map((e) => ({
        field: e.path,
        message: e.message,
      }));
      return res.status(422).json({ error: 'Validation failed', details });
    }
    res.status(500).json({ error: 'Internal server error' });
  }
};

// Formatea el documento Mongoose al contrato de API
function formatProduct(doc) {
  const components = (doc.components || []).map((component) => ({
    productId: component.productId?._id?.toString() || component.productId?.toString(),
    name: component.productId?.name,
    quantity: component.quantity,
  }));

  return {
    id: doc._id.toString(),
    name: doc.name,
    price: doc.price,
    stock: doc.type === 'COMBO' ? getComboStock(doc) : doc.stock,
    category: doc.category,
    type: doc.type || 'SIMPLE',
    components,
    createdAt: doc.createdAt,
    updatedAt: doc.updatedAt,
  };
}

function normalizeComponents(components = []) {
  const grouped = new Map();

  for (const component of components) {
    if (!component.productId || !Number.isInteger(Number(component.quantity)) || Number(component.quantity) <= 0) {
      continue;
    }

    const productId = component.productId.toString();
    grouped.set(productId, (grouped.get(productId) || 0) + Number(component.quantity));
  }

  return Array.from(grouped.entries()).map(([productId, quantity]) => ({
    productId,
    quantity,
  }));
}

function buildProductPayload(body) {
  const type = body.type || 'SIMPLE';
  const payload = {
    name: body.name,
    price: body.price,
    stock: type === 'COMBO' ? 0 : body.stock,
    category: body.category,
    type,
    components: type === 'COMBO' ? normalizeComponents(body.components) : [],
  };

  return payload;
}

function getComboStock(product) {
  if (!product.components?.length) return 0;

  const componentStocks = product.components.map((component) => {
    const componentStock = component.productId?.stock;
    if (typeof componentStock !== 'number' || component.quantity <= 0) return 0;
    return Math.floor(componentStock / component.quantity);
  });

  return Math.min(...componentStocks);
}

      copiaDescriptiva: q.copiaDescriptiva,

export const updateUser = async (req, res) => {

    const { name, username, password, role } = req.body;

    const user = await User.findById(req.params.id).select('+password');

    if (!user) {
      return res.status(404).json({
        error: 'User not found',
        id: req.params.id
      });
    }

    // 🔥 actualizar campos
    if (name) user.name = name;
    if (username) user.username = username;
    if (role) user.role = role;

    res.json(formatUser(user));

  } catch (err) {

  res.status(500).json({ error: 'Internal server error' });
};
};

import User from "../models/User.js";

// 🔐 VERIFICAR TOKEN
export const verifyToken = async (req, res, next) => {
  try {
    let token;

    if (
      req.headers.authorization &&
      req.headers.authorization.startsWith('Bearer ')
    ) {
      token = req.headers.authorization.split(' ')[1];
    }

    if (!token) {
      return res.status(401).json({
        message: 'Acceso denegado, no hay token'
      });
    }

    const decoded = jwt.verify(token, process.env.JWT_SECRET);

    // 🔥 traer usuario sin password
    const user = await User.findById(decoded.id).select('-password');

    if (!user) {
      return res.status(401).json({
        message: 'Usuario no encontrado'
      });
    }

    req.user = user;

    next();

  } catch (error) {
    return res.status(401).json({
      message: 'Token no válido o expirado'
    });
  }
};

export const rejectDangerousKeys = (req, res, next) => {
  if (hasDangerousKey(req.body) || hasDangerousKey(req.query) || hasDangerousKey(req.params)) {
    return res.status(400).json({ error: 'Invalid request payload' });
  }

  next();
};

export const loginRateLimit = (req, res, next) => {
  const windowMs = Number(process.env.LOGIN_RATE_LIMIT_WINDOW_MS) || 15 * 60 * 1000;
  const maxAttempts = Number(process.env.LOGIN_RATE_LIMIT_MAX) || 5;
  const now = Date.now();
  const username = typeof req.body?.username === 'string' ? req.body.username.toLowerCase() : '';
  const key = `${req.ip}:${username}`;
  const current = loginAttempts.get(key);

  if (!current || current.resetAt <= now) {
    loginAttempts.set(key, { count: 1, resetAt: now + windowMs });
    return next();
  }

  if (current.count >= maxAttempts) {
    return res.status(429).json({
      success: false,
      message: 'Demasiados intentos. Intenta de nuevo más tarde'
    });
  }

  current.count += 1;
  loginAttempts.set(key, current);
  next();
};

function hasDangerousKey(value) {
  if (!value || typeof value !== 'object') return false;

  for (const key of Object.keys(value)) {
    if (key.startsWith('$') || key.includes('.')) {
      return true;
    }

    if (hasDangerousKey(value[key])) {
      return true;
    }
  }

  return false;
}

const HOSTS_LOCALES = new Set(['localhost', '127.0.0.1', '::1', '[::1]']);

export function dbEsperada() {
  return (process.env.SEED_EXPECTED_DB || 'bar-db').trim();
}

export function esProduccion() {
  return String(process.env.NODE_ENV || '').trim().toLowerCase() === 'production';
}

export function redactarUri(uri) {
  try {
    const url = new URL(uri);
    if (url.username) url.username = '****';
    if (url.password) url.password = '****';
    return url.toString();
  } catch {
    return '(uri inválida)';
  }
}

export function validarUriLocal(uri) {
  if (!uri || typeof uri !== 'string' || !uri.trim()) {
    throw new Error('MONGODB_URI no definido. Abortando.');
  }

  let url;
  try {
    url = new URL(uri.trim());
  } catch {
    throw new Error(`MONGODB_URI no es una URI válida: ${redactarUri(uri)}`);
  }

  if (url.protocol !== 'mongodb:') {
    if (url.protocol === 'mongodb+srv:') {
      throw new Error('MONGODB_URI apunta a MongoDB Atlas (mongodb+srv://). Abortando: el seed SOLO trabaja contra MongoDB local.');
    }
    throw new Error(`Protocolo no soportado "${url.protocol}". Solo se permite mongodb:// local.`);
  }

  const host = url.hostname || '';
  if (!HOSTS_LOCALES.has(host.toLowerCase())) {
    throw new Error(`MONGODB_URI apunta a un host remoto ("${host}"). Abortando: el seed SOLO trabaja contra MongoDB local.`);
  }

  const db = decodeURIComponent((url.pathname || '').replace(/^\/+|\/+$/g, ''));
  const esperada = dbEsperada();
  if (!db) {
    throw new Error('MONGODB_URI no indica base de datos. Abortando.');
  }
  if (esperada && db !== esperada) {
    throw new Error(`Base de datos "${db}" no coincide con la esperada "${esperada}". Abortando.`);
  }

  return { host, db };
}

export function assertLocalOnly(uri) {
  if (esProduccion()) {
    throw new Error('NODE_ENV=production detectado. Abortando: el seed NO puede ejecutarse en producción.');
  }
  return validarUriLocal(uri);
}

export function cargarUriLocal() {
  const uri = process.env.MONGODB_URI;
  const info = assertLocalOnly(uri);
  return { uri, ...info };
}

import mongoose from 'mongoose';
import Finanza from '../models/Finanza.js';
import CierreFinanciero from '../models/CierreFinanciero.js';

// 💼 Helpers del periodo financiero del módulo Finanzas.
// Compartidos por finanzasController y gastosRecurrentesController para asignar
// cada movimiento al periodo financiero ABIERTO en el momento de su creación.

export function round2(n) {
  return Math.round((Number(n) + Number.EPSILON) * 100) / 100;
}

// 🔓 Devuelve el periodo ABIERTO actual. Si no existe (primera vez o justo
// después de un cierre), lo crea de forma atómica e idempotente:
//
//  - Continuidad exacta: fechaInicio === fechaFin del cierre más reciente.
//    Sin saltos ni solapamientos (ej. 09/10 02:15, no 09/10 00:00 ni 02:16).
//  - El índice único parcial garantiza que jamás existan dos periodos abiertos,
//    incluso si dos solicitudes lo intentan a la vez.
export async function getPeriodoAbierto() {
  const ultimoCerrado = await CierreFinanciero.findOne({ estado: 'CERRADO' })
    .sort({ fechaFin: -1 })
    .select('fechaFin')
    .lean();

  const fechaInicio = ultimoCerrado?.fechaFin || new Date();

  try {
    return await CierreFinanciero.findOneAndUpdate(
      { estado: 'ABIERTO' },
      {
        $setOnInsert: {
          estado: 'ABIERTO',
          esAbierto: true,
          fechaInicio,
        },
      },
      { upsert: true, new: true, setDefaultsOnInsert: true }
    );
  } catch (err) {
    // 🏁 Carrera de arranque: otra solicitud creó el periodo ABIERTO justo antes
    // (el índice único parcial rechazó el insert con E11000). Devolvemos el existente.
    if (err.code !== 11000) throw err;
    const existente = await CierreFinanciero.findOne({ estado: 'ABIERTO' });
    if (existente) return existente;
    throw err;
  }
}

// 🧮 Totales LIVE del periodo: suma los movimientos asociados por
// cierreFinancieroId. Fórmula: RESULTADO = INGRESOS - GASTOS - INVERSIONES.
// Todo se calcula por backend (nunca se confía en totales del cliente).
export async function calcularTotalesPeriodo(cierreId) {
  const agrupado = await Finanza.aggregate([
    { $match: { cierreFinancieroId: new mongoose.Types.ObjectId(String(cierreId)) } },
    {
      $group: {
        _id: '$tipo',
        monto: { $sum: '$monto' },
        n: { $sum: 1 },
      },
    },
  ]);

  const resumen = { ingresos: 0, gastos: 0, inversiones: 0, resultado: 0, cantidadMovimientos: 0 };

  for (const g of agrupado) {
    resumen.cantidadMovimientos += g.n;
    const monto = round2(g.monto);
    if (g._id === 'INGRESO') resumen.ingresos = monto;
    else if (g._id === 'GASTO') resumen.gastos = monto;
    else if (g._id === 'INVERSION') resumen.inversiones = monto;
  }

  resumen.resultado = round2(resumen.ingresos - resumen.gastos - resumen.inversiones);
  return resumen;
}
